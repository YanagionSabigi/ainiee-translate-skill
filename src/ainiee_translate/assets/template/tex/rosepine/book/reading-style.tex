% Dawn screen default. Define \WhitePaper before main.tex for white paper.
% Dawn Overlay is warmer and darker than the page, avoiding a white patch.
\definecolor{rpsurface}{HTML}{f2e9e1}
\ifdefined\WhitePaper
    \colorlet{rppage}{white}
    \colorlet{rpsurface}{proofcolor!3!white}
\else
    \colorlet{rppage}{rpbase}
\fi
\pagecolor{rppage}
\color{rptext}
\newcounter{theorem}[chapter]
\renewcommand{\thetheorem}{\thechapter.\arabic{theorem}}
\providecommand{\theHtheorem}{\theHchapter.\arabic{theorem}}
\newcommand{\statementtitle}[2]{%
    #1~\thetheorem\IfBlankTF{#2}{}{\normalfont\ (#2)}}
\tcbset{
    reading/.style={enhanced jigsaw,breakable,arc=0pt,
        frame hidden,boxrule=0pt,left=3mm,right=3mm,
        before skip=9pt,after skip=9pt,top=2pt,bottom=3pt,
        coltext=rptext,colbacktitle=rppage,
        coltitle=accent,fonttitle=\bfseries,
        before upper={\phantomsection},
        toptitle=3pt,bottomtitle=3pt},
    quietline/.style={reading,colback=rppage}
}
\newtcolorbox{statementbox}[2]{reading,title={#1},
    colback=#2,colbacktitle=#2}
\newcommand{\DeclareStatement}[3]{%
    \NewDocumentEnvironment{#1}{O{}}{%
        \refstepcounter{theorem}%
        \begin{statementbox}{\statementtitle{#2}{##1}}{#3}%
    }{\end{statementbox}}}
\DeclareStatement{theorem}{定理}{rpsurface}
\DeclareStatement{corollary}{推论}{rpsurface}
\DeclareStatement{proposition}{命题}{rppage}
\DeclareStatement{lemma}{引理}{rppage}
\DeclareStatement{definition}{定义}{rppage}
\DeclareStatement{axiom}{公理}{rppage}
\DeclareStatement{claim}{声明}{rppage}
\newcommand{\termcn}[1]{{\kaishu #1}}
\newtcolorbox{proofbox}{quietline,
    borderline west={0.6pt}{0pt}{proofcolor!60!rppage}}
\renewenvironment{proof}[1][证明]{%
    \begin{proofbox}\pushQED{\qed}%
    {\bfseries\color{proofcolor}#1.}\par\nobreak
}{\par\nobreak\popQED\end{proofbox}}
\renewcommand{\qedsymbol}{\ensuremath{\square}}
\newenvironment{quotationbox}{%
    \begin{tcolorbox}[quietline,
        borderline west={0.6pt}{0pt}{quotecolor!65!rppage}]
    \small\kaishu
}{\end{tcolorbox}}
\newcommand{\quoteauthor}[2]{%
    \par\nobreak\begin{flushright}—— #1《#2》\end{flushright}}
\newtcolorbox{illustrationbox}[2]{quietline,
    coltitle=#2,colbacktitle=rppage,title={#1},
    borderline west={0.4pt}{0pt}{#2!40!rppage}}
\newcommand{\DeclareIllustration}[3]{%
    \NewDocumentEnvironment{#1}{O{}}{%
        \refstepcounter{theorem}%
        \begin{illustrationbox}{\statementtitle{#2}{##1}}{#3}%
    }{\end{illustrationbox}}}
\DeclareIllustration{example}{例}{examplecolor}
\DeclareIllustration{exercise}{习题}{exercisecolor}
